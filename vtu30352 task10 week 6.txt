import java.io.*;

public class Solution {

    // Find longest palindromic substring length
    public static int longestPalindrome(String s) {

        // Transform string:
        // "abba" -> "#a#b#b#a#"
        StringBuilder t = new StringBuilder("^");

        for (int i = 0; i < s.length(); i++) {
            t.append("#");
            t.append(s.charAt(i));
        }

        t.append("#$");

        int n = t.length();
        int[] p = new int[n];

        int center = 0;
        int right = 0;
        int maxLength = 0;

        for (int i = 1; i < n - 1; i++) {

            int mirror = 2 * center - i;

            if (i < right) {
                p[i] = Math.min(right - i, p[mirror]);
            }

            // Expand around center
            while (t.charAt(i + (1 + p[i]))
                    == t.charAt(i - (1 + p[i]))) {
                p[i]++;
            }

            // Update center and right boundary
            if (i + p[i] > right) {
                center = i;
                right = i + p[i];
            }

            maxLength = Math.max(maxLength, p[i]);
        }

        return maxLength;
    }

    public static void main(String[] args) throws IOException {

        BufferedReader br =
            new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());

        String s = br.readLine();

        // Generate every rotation
        for (int i = 0; i < n; i++) {

            String rotation = s.substring(i) + s.substring(0, i);

            int answer = longestPalindrome(rotation);

            System.out.println(answer);
        }
    }
}
