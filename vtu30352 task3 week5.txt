import java.io.*;
import java.util.*;

public class Solution {

    static int[] maxSubarray(List<Integer> arr) {
        // Maximum subarray sum
        int currentSum = arr.get(0);
        int maxSubarray = arr.get(0);
        
        // Maximum subsequence sum
        int positiveSum = 0;
        int largest = arr.get(0);
        
        for (int i = 0; i < arr.size(); i++) {
            int value = arr.get(i);
            
            // -------- Maximum Subarray (Kadane's Algorithm) --------
            if (i > 0) {
                currentSum = Math.max(value, currentSum + value);
                maxSubarray = Math.max(maxSubarray, currentSum);
            }
            
            // -------- Maximum Subsequence --------
            if (value > 0) {
                positiveSum += value;
            }
            
            // Find the largest element
            largest = Math.max(largest, value);
        }
        
        // If all elements are negative, we must choose one element
        int maxSubsequence;
        if (positiveSum == 0) {
            maxSubsequence = largest;
        } else {
            maxSubsequence = positiveSum;
        }
        
        return new int[]{maxSubarray, maxSubsequence};
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        if (!scanner.hasNextInt()) return;
        
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            List<Integer> arr = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                arr.add(scanner.nextInt());
            }
            
            int[] result = maxSubarray(arr);
            System.out.println(result[0] + " " + result[1]);
        }
        scanner.close();
    }
}
